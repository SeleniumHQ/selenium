def initialize(args: nil, **)
  args = Array(args.dup)
  unless args.any? { |arg| arg.include?('--connect-existing') || arg.include?('--websocket-port') }
    args << '--websocket-port'
    args << '0'
  end

  configure_debug_args(args) if ENV.key?('SE_DEBUG')

  super
end
