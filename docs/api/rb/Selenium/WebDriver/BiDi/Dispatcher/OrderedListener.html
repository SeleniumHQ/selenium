Invoked with one payload at a time, in arrival order, on its own worker thread. Used only when SE_ORDERED_EVENTS is true; kept for evaluating ordered delivery.

def initialize(...)
  super
  @queue = Queue.new
  @worker = Thread.new do
    until (payload = @queue.pop).equal?(STOP)
      invoke(payload)
    end
  end
end
