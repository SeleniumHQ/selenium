  theme:,
  contexts: Serialization::UNSET,
  user_contexts: Serialization::UNSET
)

    contexts: contexts,
    user_contexts: user_contexts
  )
  execute(cmd: 'emulation.setForcedColorsModeThemeOverride', params: params)
end
