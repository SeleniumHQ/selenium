def start(
  active_context: Serialization::UNSET,
  preset: Serialization::UNSET,
  entries: Serialization::UNSET,
  interval: Serialization::UNSET,
  features: Serialization::UNSET,
  threads: Serialization::UNSET
)
  params = StartParameters.build(
    active_context: active_context,
    preset: preset,
    entries: entries,
    interval: interval,
    features: features,
    threads: threads
  )
  execute(cmd: 'moz:profiler.start', params: params)
end
