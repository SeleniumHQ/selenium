def initialize(args: nil, **)
  if ENV.key?('SE_DEBUG')
    args = Array(args.dup)
    warn_driver_log_override if args.reject! { |arg| arg.include?('log-level') || arg.include?('silent') }
    args << '--log-level=DEBUG'
  end

  super
end
